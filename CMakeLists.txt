cmake_minimum_required(VERSION 3.25)
project(KEYRecompProbe LANGUAGES CXX)
option(KEY_ENABLE_GRAPHICS "Build the native Aurora graphics bridge" OFF)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
enable_testing()
add_library(key_memory runtime/src/memory.cpp runtime/src/clock.cpp runtime/src/disc.cpp)
target_include_directories(key_memory PUBLIC runtime/include)
if(KEY_AX_SOURCES)
  target_sources(key_memory PRIVATE "${KEY_AX_SOURCES}/ax_processor.cpp" runtime/src/ax.cpp)
  target_compile_definitions(key_memory PRIVATE KEY_WITH_AX=1)
  target_include_directories(key_memory PRIVATE runtime/include/key references/WiiCompiled/runtime/include "${KEY_AX_SOURCES}")
  add_executable(ax_tests tests/ax_tests.cpp)
  target_include_directories(ax_tests PRIVATE runtime/include/key references/WiiCompiled/runtime/include "${KEY_AX_SOURCES}")
  target_link_libraries(ax_tests PRIVATE key_memory)
  add_test(NAME ax_command_processing COMMAND ax_tests)
  if(MINGW)
    target_link_options(ax_tests PRIVATE -static)
  endif()
endif()
add_executable(memory_tests tests/memory_tests.cpp)
target_link_libraries(memory_tests PRIVATE key_memory)
add_test(NAME checked_memory COMMAND memory_tests)
add_executable(device_tests tests/device_tests.cpp)
target_link_libraries(device_tests PRIVATE key_memory)
add_test(NAME device_contracts COMMAND device_tests)
add_executable(audio_dma_tests tests/audio_dma_tests.cpp)
target_link_libraries(audio_dma_tests PRIVATE key_memory)
add_test(NAME audio_dma_contracts COMMAND audio_dma_tests)
add_executable(host_timeline_tests tests/host_timeline_tests.cpp)
target_link_libraries(host_timeline_tests PRIVATE key_memory)
add_test(NAME host_backpressure_clock COMMAND host_timeline_tests)
add_executable(gx_bridge_tests tests/gx_bridge_tests.cpp renderer/command_bridge.cpp)
target_include_directories(gx_bridge_tests PRIVATE renderer)
add_test(NAME gx_stream_contracts COMMAND gx_bridge_tests)
add_executable(gpu_pending_tests tests/gpu_pending_tests.cpp)
target_include_directories(gpu_pending_tests PRIVATE runtime/include)
add_test(NAME gpu_copy_hazards COMMAND gpu_pending_tests)
if(KEY_ENABLE_GRAPHICS)
  add_subdirectory(renderer)
endif()
if(KEY_PROBE_SOURCES)
  if(NOT WIN32 OR NOT CMAKE_SIZEOF_VOID_P EQUAL 8 OR NOT CMAKE_CXX_COMPILER_ID MATCHES "Clang")
    message(FATAL_ERROR "The boot probe currently requires Windows x64 Clang")
  endif()
  include("${KEY_PROBE_SOURCES}")
  add_library(key_isa runtime/src/quantized.cpp runtime/third_party/wiicompiled/ppc_quantized.cpp)
  target_include_directories(key_isa PUBLIC runtime/include references/WiiCompiled/runtime/include)
  target_link_libraries(key_isa PUBLIC key_memory)
  target_compile_options(key_isa PUBLIC -march=x86-64-v2 -ffp-contract=off)
  add_executable(key-boot-probe runtime/src/probe.cpp runtime/src/helpers.cpp runtime/src/report.cpp runtime/src/ios.cpp runtime/src/gx_stream.cpp runtime/src/scheduler.cpp ${KEY_GENERATED_SOURCES})
  target_include_directories(key-boot-probe PRIVATE runtime/include "${KEY_REGISTRY_DIR}" references/WiiCompiled/runtime/include)
  target_link_libraries(key-boot-probe PRIVATE key_isa bcrypt user32)
  target_compile_options(key-boot-probe PRIVATE -ffp-contract=off)
  if(KEY_AX_SOURCES)
    target_compile_definitions(key-boot-probe PRIVATE KEY_WITH_AX=1)
  endif()
  target_precompile_headers(key-boot-probe PRIVATE runtime/include/guest_abi/guest_abi.h)
  if(KEY_ENABLE_GRAPHICS)
    target_link_libraries(key-boot-probe PRIVATE key_graphics)
    target_compile_definitions(key-boot-probe PRIVATE KEY_WITH_GRAPHICS=1)
  endif()
  add_executable(helper_tests tests/helper_tests.cpp runtime/src/helpers.cpp)
  target_include_directories(helper_tests PRIVATE runtime/include references/WiiCompiled/runtime/include)
  target_link_libraries(helper_tests PRIVATE key_isa)
  add_test(NAME startup_helpers COMMAND helper_tests)
  add_executable(report_tests tests/report_tests.cpp runtime/src/report.cpp)
  target_include_directories(report_tests PRIVATE runtime/include references/WiiCompiled/runtime/include)
  target_link_libraries(report_tests PRIVATE key_memory)
  add_test(NAME report_formatting COMMAND report_tests)
  add_executable(scheduler_tests tests/scheduler_tests.cpp runtime/src/scheduler.cpp)
  target_link_libraries(scheduler_tests PRIVATE key_isa)
  add_test(NAME native_continuations COMMAND scheduler_tests)
  add_executable(ios_tests tests/ios_tests.cpp runtime/src/ios.cpp)
  target_include_directories(ios_tests PRIVATE runtime/include references/WiiCompiled/runtime/include)
  target_link_libraries(ios_tests PRIVATE key_memory)
  add_test(NAME native_nand_contracts COMMAND ios_tests)
  set_tests_properties(native_nand_contracts PROPERTIES WORKING_DIRECTORY "${CMAKE_CURRENT_SOURCE_DIR}")
endif()
if(MINGW)
  add_link_options(-static)
  target_link_options(memory_tests PRIVATE -static)
  target_link_options(device_tests PRIVATE -static)
  target_link_options(audio_dma_tests PRIVATE -static)
  target_link_options(host_timeline_tests PRIVATE -static)
  target_link_options(gx_bridge_tests PRIVATE -static)
  target_link_options(gpu_pending_tests PRIVATE -static)
  if(TARGET key-boot-probe)
    target_link_options(key-boot-probe PRIVATE -static -mwindows)
    target_link_options(helper_tests PRIVATE -static)
    target_link_options(report_tests PRIVATE -static)
    target_link_options(scheduler_tests PRIVATE -static)
    target_link_options(ios_tests PRIVATE -static)
  endif()
endif()
