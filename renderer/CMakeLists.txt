set(_key_root "${CMAKE_CURRENT_LIST_DIR}/..")
set(AURORA_ENABLE_TESTS OFF CACHE BOOL "" FORCE)
set(AURORA_ENABLE_EXAMPLES OFF CACHE BOOL "" FORCE)
set(AURORA_ENABLE_DVD OFF CACHE BOOL "" FORCE)
set(AURORA_ENABLE_CARD OFF CACHE BOOL "" FORCE)
set(AURORA_ENABLE_THP OFF CACHE BOOL "" FORCE)
set(AURORA_ENABLE_RMLUI OFF CACHE BOOL "" FORCE)
set(AURORA_TARGET_RVL OFF CACHE BOOL "" FORCE)
set(AURORA_OPTIMIZE_GX_DEBUG ON CACHE BOOL "" FORCE)
set(AURORA_DAWN_PROVIDER package CACHE STRING "" FORCE)
set(AURORA_SDL3_PROVIDER package CACHE STRING "" FORCE)
set(AURORA_USE_SYSTEM_DEPENDENCIES OFF CACHE BOOL "" FORCE)
foreach(_dep IN ITEMS dawn_prebuilt sdl3_prebuilt abseil-cpp fmt freetype imgui png sqlite3 tracy xxhash zlib zstd)
  if(EXISTS "${_key_root}/private/build/aurora/_deps/${_dep}-src")
    string(TOUPPER "${_dep}" _upper)
    set("FETCHCONTENT_SOURCE_DIR_${_upper}" "${_key_root}/private/build/aurora/_deps/${_dep}-src" CACHE PATH "Reuse pinned local dependency sources" FORCE)
  endif()
endforeach()
add_subdirectory("${_key_root}/references/WiiCompiled/aurora-main" aurora)
add_library(key_graphics backend.cpp command_bridge.cpp audio.cpp)
target_include_directories(key_graphics PUBLIC "${_key_root}/runtime/include" "${CMAKE_CURRENT_LIST_DIR}" PRIVATE "${_key_root}/references/WiiCompiled/aurora-main/lib")
target_include_directories(key_graphics PRIVATE "${_key_root}/references/WiiCompiled/runtime/include")
target_link_libraries(key_graphics PUBLIC key_memory aurora::core aurora::gx aurora::vi aurora::wpad)
target_link_libraries(key_graphics PRIVATE absl::flat_hash_map key_controller)
add_executable(graphics_smoke smoke.cpp)
target_link_libraries(graphics_smoke PRIVATE key_graphics)
add_executable(audio_output_tests "${_key_root}/tests/audio_output_tests.cpp" audio.cpp)
target_include_directories(audio_output_tests PRIVATE "${CMAKE_CURRENT_LIST_DIR}")
target_link_libraries(audio_output_tests PRIVATE key_memory SDL3::SDL3)
add_custom_command(TARGET audio_output_tests POST_BUILD
  COMMAND "${CMAKE_COMMAND}" -E copy_if_different "$<TARGET_FILE:SDL3::SDL3>" "$<TARGET_FILE_DIR:audio_output_tests>")
if(MINGW)
  target_link_options(audio_output_tests PRIVATE -static)
endif()
add_test(NAME pcm_output_backpressure COMMAND audio_output_tests)
set_tests_properties(pcm_output_backpressure PROPERTIES ENVIRONMENT "SDL_AUDIODRIVER=dummy")
